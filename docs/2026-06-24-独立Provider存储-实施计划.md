**Goal:** 把 Claude（`app_type` 为 `claude` / `claude-desktop`）供应商从 cc-switch 的 SQLite 库迁到 monitor 自己的库 `~/.cc-switch-monitor/providers.db`，monitor 不再依赖 cc-switch 即可管理 Claude Code 的 API 配置。

**Architecture:** 路线 A —— 按 `app_type` 路由到不同库。新增 `OWN_DB_PATH` 与 `_db_path_for(app_type)` 帮助函数；所有 Claude DB 路径替换为 `OWN_DB_PATH`，Codex 路径不变。首次启动自动从 cc-switch 导入 Claude 供应商（幂等，用 `.imported` 标记文件防重复）。当前供应商标记由读 `~/.claude/settings.json` 比对得出，不再信任/写入库内 `is_current`。

**Tech Stack:** Python 3.11（标准库 `sqlite3` / `json` / `os` / `uuid` / `tempfile` / `unittest`）；`tkinter` UI；无第三方依赖（TOML 读取已有 `tomllib`）。

## Global Constraints

- 仅迁移 Claude，Codex 完全不变（仍读写 `~/.cc-switch/cc-switch.db` 与 `~/.codex/`）。
- 自有库 schema 与 cc-switch 的 `providers` / `provider_endpoints` 两表一致（便于整行 INSERT 迁移）。
- 不建 `provider_health` / `stream_check_logs` 表；`delete_provider` Claude 分支对这两表的 DELETE 用 `try/except sqlite3.OperationalError: pass` 包裹。

```python
def import_claude_from_ccswitch():
    """从 cc-switch 库一次性导入 Claude 供应商到自有库。幂等（INSERT OR IGNORE）。
    完成后写标记文件，防止重启后误重新导入用户已清空的列表。
    返回 (导入数, 失败数)。
    """
    if not os.path.exists(DB_PATH):
        return 0, 0
    imported, failed = 0, 0
    try:
        src = sqlite3.connect(f"file:{DB_PATH}?mode=ro", uri=True)
        src.row_factory = sqlite3.Row
        rows = src.execute(
            """SELECT id, app_type, name, settings_config, website_url, category,
                      created_at, notes, meta, is_current, in_failover_queue,
                      provider_type, sort_index
               FROM providers
               WHERE app_type IN ('claude', 'claude-desktop')"""
        ).fetchall()
        dst = sqlite3.connect(OWN_DB_PATH, timeout=10)
        for r in rows:
            try:
                dst.execute(
                    """INSERT OR IGNORE INTO providers
                       (id, app_type, name, settings_config, website_url, category,
                        created_at, notes, meta, is_current, in_failover_queue,
                        provider_type, sort_index)
                       VALUES (?,?,?,?,?,?,?,?,?,?,?,?,?)""",
                    (r["id"], r["app_type"], r["name"], r["settings_config"],
                     r["website_url"], r["category"], r["created_at"], r["notes"],
                     r["meta"], r["is_current"], r["in_failover_queue"],
                     r["provider_type"], r["sort_index"]),
                )
                if dst.total_changes:
                    imported += 1
                # 端点
                eps = src.execute(
                    "SELECT provider_id, app_type, url, added_at FROM provider_endpoints WHERE provider_id=? AND app_type=?",
                    (r["id"], r["app_type"]),
                ).fetchall()
                for ep in eps:
                    dst.execute(
                        "INSERT OR IGNORE INTO provider_endpoints (provider_id, app_type, url, added_at) VALUES (?,?,?,?)",
                        (ep["provider_id"], ep["app_type"], ep["url"], ep["added_at"]),
                    )
            except Exception:
                failed += 1
        dst.commit()
        dst.close()
        src.close()
    except Exception:
        return 0, 0
    # 写标记文件
    try:
        with open(IMPORTED_MARKER_PATH, "w", encoding="utf-8") as f:
            f.write("1")
    except Exception:
        pass
    audit_log("IMPORT", "", "", f"n={imported} failed={failed} from=cc-switch")
    return imported, failed
```

```python
                    """INSERT OR IGNORE INTO providers ... VALUES (...)""", (...)
                )

                    imported += 1
```

```python
        p["is_current"] = False

        p_key = p.get("api_key", "")

            if cur_key and p_key == cur_key:
                p["is_current"] = True
            else:
                p["is_current"] = True

                warnings.append("当前 settings.json 的 key 与本供应商不匹配")

    return providers
```

```python
def _change_token():
    """返回一个变更令牌：own_db / cc_switch_db / settings.json 三者 mtime 的最大值。
    任一文件缺失记 0。用于 _poll_data 的变化检测：三个来源任一变化都应触发刷新。

    for p in (OWN_DB_PATH, DB_PATH, SETTINGS_PATH):
        try:
            token = max(token, os.path.getmtime(p))
        except Exception:
            pass
    return token
```

```python
def maybe_auto_import_on_startup():
    """启动时调用：若无 .imported 标记且自有库无 Claude 行，从 cc-switch 导入一次。
    返回导入条数。有标记或已导入过则跳过。
    """
    if os.path.exists(IMPORTED_MARKER_PATH):
        return 0
    # 检查自有库是否已有 Claude 行
    try:
        conn = sqlite3.connect(OWN_DB_PATH, timeout=10)
        n = conn.execute(
            "SELECT COUNT(*) FROM providers WHERE app_type IN ('claude','claude-desktop')"
        ).fetchone()[0]
        conn.close()
    except Exception:
        n = 0
    if n > 0:
        # 已有数据但无标记：补标记，不导入
        try:
            with open(IMPORTED_MARKER_PATH, "w", encoding="utf-8") as f:
                f.write("1")
        except Exception:
            pass
        return 0
    imported, _failed = import_claude_from_ccswitch()
    return imported
```
