def calculate_table_column_widths(total_width):
    cols = ("name", "app", "role", "endpoint", "key", "latency", "detail")
    minimum = {
        "name": 110,
        "app": 56,
        "role": 76,
        "endpoint": 120,
        "key": 96,
        "latency": 66,
        "detail": 120,
    }
    weights = {
        "name": 2,
        "endpoint": 3,
        "detail": 3,
    }
    width = max(int(total_width or 0), sum(minimum.values()))
    result = dict(minimum)
    extra = width - sum(minimum.values())
    if extra > 0:
        weight_total = sum(weights.values())
        assigned = 0
        weighted_cols = list(weights)
        for col in weighted_cols[:-1]:
            add = int(extra * weights[col] / weight_total)
            result[col] += add
            assigned += add
        result[weighted_cols[-1]] += extra - assigned
    return {col: result[col] for col in cols}
```

def _schedule_table_layout(self, event=None):
    try:
        if getattr(self, "_table_layout_after", None):
            self.after_cancel(self._table_layout_after)
        self._table_layout_after = self.after(60, self._apply_table_layout)
    except Exception:
        pass

def _apply_table_layout(self):
    self._table_layout_after = None
    try:
        width = self.tree.winfo_width()
        if width <= 1:
            return
        widths = calculate_table_column_widths(width)
        for col, col_width in widths.items():
            self.tree.column(col, width=col_width, minwidth=col_width)
    except Exception:
        pass
```
